\begin{abstract}
This document introduces the charging behaviour of a
resistor-capacitor (RC) network and derives the capacitor
voltage at one time constant.
\end{abstract}

\section{Introduction}
When a capacitor $C$ is charged through a series resistor
$R$ from a constant supply $V_0$, the capacitor voltage
rises according to

\begin{equation}
V(t) = V_0\left(1-e^{-t/RC}\right).
\end{equation}

The product $RC$ has units of seconds and is called the
time constant, $\tau=RC$



\section{Voltage at the Time Constant}

\begin{equation}
\begin{aligned}
V(\tau)
&= V_0\left(1-e^{-\tau/RC}\right) \\
&= V_0\left(1-e^{-RC/RC}\right)
&& \text{since } \tau = RC \\
&= V_0\left(1-e^{-1}\right) \\
&\approx V_0(1-0.368) \\
&\approx 0.632V_0
\end{aligned}
\end{equation}
